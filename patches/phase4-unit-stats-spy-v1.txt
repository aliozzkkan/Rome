
  // === PHASE 4: Attack / Defense / Speed stats + upgrade cards + Spy ===
  const BASE_UNIT_STATS = {
    legion:  { attack:1, defense:2, speed:1 },
    cavalry: { attack:2, defense:1, speed:2 },
  };

  function cloneBaseUnitStats() {
    return {
      legion:{...BASE_UNIT_STATS.legion},
      cavalry:{...BASE_UNIT_STATS.cavalry},
    };
  }

  function ensurePlayerMilitaryData(player) {
    if (!player) return;
    if (!player.unitStats) player.unitStats = cloneBaseUnitStats();
    if (!(player.spyIntel instanceof Set)) player.spyIntel = new Set(player.spyIntel || []);
  }

  function statsForPlayer(playerId,type) {
    if (!playerId) return BASE_UNIT_STATS[type] || BASE_UNIT_STATS.legion;
    const p = playerById(playerId);
    ensurePlayerMilitaryData(p);
    return p?.unitStats?.[type] || BASE_UNIT_STATS[type];
  }

  function selectedArmySpeed(playerId,legions,cavalry) {
    const speeds = [];
    if (legions > 0) speeds.push(statsForPlayer(playerId,'legion').speed);
    if (cavalry > 0) speeds.push(statsForPlayer(playerId,'cavalry').speed);
    return speeds.length ? Math.max(1,Math.min(...speeds)) : 0;
  }

  function attackPowerFor(playerId,legions,cavalry) {
    const l = statsForPlayer(playerId,'legion');
    const c = statsForPlayer(playerId,'cavalry');
    return legions*l.attack + cavalry*c.attack;
  }

  function defensePowerFor(playerId,g) {
    if (!g) return 0;
    const l = statsForPlayer(playerId,'legion');
    const c = statsForPlayer(playerId,'cavalry');
    return legionTotalInGroup(g)*l.defense + cavalryTotalInGroup(g)*c.defense;
  }

  function humanCanSeeMilitaryStats(playerId) {
    if (!playerId || playerId === HUMAN_PLAYER_ID) return true;
    const human = playerById(HUMAN_PLAYER_ID);
    ensurePlayerMilitaryData(human);
    return human.spyIntel.has(playerId);
  }

  const baseStatsBuildDeckAndHands = buildDeckAndHands;
  buildDeckAndHands = function(...args) {
    baseStatsBuildDeckAndHands.apply(this,args);
    const deck = [];
    const push = (type,n) => { for(let i=0;i<n;i++) deck.push(card(type)); };
    push('road',25); push('fort',20); push('legion',20); push('reinforcement',8);
    push('attack_upgrade',7); push('defense_upgrade',6); push('speed_upgrade',5); push('spy',4); push('siege',5);
    if (deck.length !== TOTAL_CARDS) throw new Error(`Yeni deste 100 kart olmalı, ${deck.length} oldu.`);
    game.deck = seededShuffle(deck);
    game.spentPile = [];
    for (const p of game.players) {
      p.hand = [];
      p.discard = [];
      if (p.romeCoins == null) p.romeCoins = 0;
      p.lastRomeIncome = 0;
      p.unitStats = cloneBaseUnitStats();
      p.spyIntel = new Set();
      for(let i=0;i<7;i++) p.hand.push(game.deck.pop());
    }
    if (typeof renderReferenceTable === 'function') renderReferenceTable();
  };

  function spendSpecificHandCard(cardId) {
    const p = currentPlayer();
    const idx = p.hand.findIndex(c=>c.id===cardId);
    if (idx < 0) return null;
    const [c] = p.hand.splice(idx,1);
    game.spentPile.push(c);
    game.cardsPlayedThisTurn += 1;
    game.selectedCardId = null;
    return c;
  }

  const baseStatsSelectCard = selectCard;
  selectCard = function(cardId) {
    const c = currentPlayer()?.hand?.find(x=>x.id===cardId);
    if (c && ['attack_upgrade','defense_upgrade','speed_upgrade','spy'].includes(c.type)) {
      if (!canPlayAnotherCard()) { toast('Bu tur 3 kart sınırına ulaştın.',false); return; }
      if (currentPlayer().hand.length<=1) { toast('Tur sonunda atmak için 1 kart bırakmalısın.',false); return; }
      if (c.type==='spy') openSpyModal(c);
      else openPermanentUpgradeModal(c,c.type==='attack_upgrade'?'attack':c.type==='defense_upgrade'?'defense':'speed');
      return;
    }
    return baseStatsSelectCard.call(this,cardId);
  };

  // Movement now follows upgraded Speed values.
  updateMixedPlannerReachable = function() {
    if (!touchArmyPlan) return;
    const l = touchArmyPlan.legionCount || 0;
    const c = touchArmyPlan.cavalryCount || 0;
    const speed = selectedArmySpeed(game.current,l,c);
    game.reachable = speed ? new Set(completedReachableWithin(touchArmyPlan.sourceId,speed).keys()) : new Set();
    if (touchArmyPlan.targetId && !game.reachable.has(touchArmyPlan.targetId)) touchArmyPlan.targetId=null;
  };

  moveMixedFriendly = function(sourceId,targetId,legions,cavalry) {
    if (typeof autoMarchExecuting !== 'undefined' && !autoMarchExecuting && typeof cancelOrdersFromLocation === 'function') cancelOrdersFromLocation(sourceId,game.current,true);
    const sg=troopGroup(sourceId); if(!sg||sg.owner!==game.current) return false;
    if((sg.ready||0)<legions||cavalryReady(sg)<cavalry) return false;
    const speed=selectedArmySpeed(game.current,legions,cavalry);
    const dist=movementDistance(sourceId,targetId,speed);
    if(!Number.isFinite(dist)||dist>speed){toast(`Bu birlik grubunun Hız değeri ${speed}.`,false);return false;}
    if(!mixedMoveCapacityOkay(targetId,legions,cavalry)){toast(`${locationLabel(targetId)} kapasitesi yetersiz.`,false);return false;}
    removeTroops(sourceId,legions,0); removeCavalry(sourceId,cavalry,0);
    animateMixedMarch(game.current,legions,cavalry,sourceId,targetId,'move');
    let tg=troopGroup(targetId);
    if(!tg){tg={owner:game.current,ready:0,spent:0,cavReady:0,cavSpent:0};troopGroups.set(targetId,tg);}
    if(tg.owner!==game.current) return false;
    tg.spent=(tg.spent||0)+legions; tg.cavSpent=cavalrySpent(tg)+cavalry;
    const node=nodes.get(targetId); if(node&&node.type!=='home'&&(node.owner===0||node.owner===game.current)) node.owner=game.current;
    syncTroopVisual(sourceId); syncTroopVisual(targetId);
    addLog(`${legions?`${legions} Lejyon`:''}${legions&&cavalry?' + ':''}${cavalry?`${cavalry} Atlı`:''} ${dist} kart ilerledi (Hız ${speed}).`,PLAYER_NAMES[game.current]);
    return true;
  };

  // Exact stat-based combat resolver.
  resolveMixedBattle = function(sourceId,targetId,legions,cavalry,atkSupport='none',defSupport='none') {
    if (typeof autoMarchExecuting !== 'undefined' && !autoMarchExecuting && typeof cancelOrdersFromLocation === 'function') cancelOrdersFromLocation(sourceId,game.current,true);
    const attacker=game.current, source=troopGroup(sourceId), target=troopGroup(targetId), node=nodes.get(targetId);
    if(!source||source.owner!==attacker||(source.ready||0)<legions||cavalryReady(source)<cavalry) return;
    const speed=selectedArmySpeed(attacker,legions,cavalry), dist=movementDistance(sourceId,targetId,speed);
    if(!Number.isFinite(dist)||dist>speed) return;
    const defenderOwner=target?.owner||node?.owner||0;
    const oldOwner=node?.owner||target?.owner||0;
    const atkSupportPower=supportPower(atkSupport,'atk',node);
    if(atkSupportPower&&atkSupport!=='none') removeOneSupportCard(attacker,atkSupport,true);
    const attackUnits=legions+cavalry;
    const defenderUnits=unitTotalInGroup(target);
    const atkPower=attackPowerFor(attacker,legions,cavalry)+atkSupportPower;
    const defPower=defensePowerFor(defenderOwner,target)+(node?defenseBonus(node):0);
    const defenderBefore=defenderUnits;
    removeTroops(sourceId,legions,0); removeCavalry(sourceId,cavalry,0);
    animateMixedMarch(attacker,legions,cavalry,sourceId,targetId,'battle');
    let attackerWon=atkPower>defPower, attackerSurvivors=0, attackerLosses=attackUnits, defenderAfter=defenderUnits, defenderLosses=0, captured=false;
    if(attackerWon){
      let casualties=Math.min(Math.max(0,attackUnits-1),Math.floor(defPower/2));
      let surviveLegion=legions, surviveCavalry=cavalry;
      const lLoss=Math.min(surviveLegion,casualties); surviveLegion-=lLoss; casualties-=lLoss;
      surviveCavalry=Math.max(0,surviveCavalry-casualties);
      if(surviveLegion+surviveCavalry<1){ if(cavalry>0) surviveCavalry=1; else surviveLegion=1; }
      attackerSurvivors=surviveLegion+surviveCavalry; attackerLosses=attackUnits-attackerSurvivors; defenderLosses=defenderUnits; defenderAfter=0;
      troopGroups.delete(targetId);
      if(node?.type==='home'){
        const homeKeep=Math.max(1,Math.ceil(defenderUnits/2));
        troopGroups.set(targetId,{owner:oldOwner,ready:homeKeep,spent:0,cavReady:0,cavSpent:0});
        defenderAfter=homeKeep; defenderLosses=Math.max(0,defenderUnits-homeKeep); captured=false;
        let back=troopGroup(sourceId); if(!back){back={owner:attacker,ready:0,spent:0,cavReady:0,cavSpent:0};troopGroups.set(sourceId,back);}
        back.spent+=surviveLegion; back.cavSpent=cavalrySpent(back)+surviveCavalry;
        stealRandomCard(oldOwner,attacker);
      } else {
        const cap=node&&Number.isFinite(nodeCapacity(node))?nodeCapacity(node):Infinity;
        let allowed=Math.min(attackerSurvivors,cap);
        const placeCav=Math.min(surviveCavalry,allowed); allowed-=placeCav; const placeLeg=Math.min(surviveLegion,allowed);
        troopGroups.set(targetId,{owner:attacker,ready:0,spent:placeLeg,cavReady:0,cavSpent:placeCav});
        attackerSurvivors=placeLeg+placeCav;
        if(node){node.owner=attacker;captured=oldOwner!==attacker;}
      }
      addLog(node?.type==='rome'?`${locationLabel(targetId)} ele geçirildi!`:'savaşı kazandı.',PLAYER_NAMES[attacker]);
    } else {
      defenderLosses=Math.min(defenderUnits,Math.floor(atkPower/2));
      if(target&&defenderLosses>0){removeDefenderUnits(target,defenderLosses);if(unitTotalInGroup(target)<=0)troopGroups.delete(targetId);}
      defenderAfter=Math.max(0,defenderUnits-defenderLosses);
      addLog(`saldırıda durduruldu (${atkPower} vs ${defPower}).`,PLAYER_NAMES[attacker]);
    }
    syncTroopVisual(sourceId); syncTroopVisual(targetId); renderAll();
    if(typeof queueBattleReport==='function') queueBattleReport({
      attacker,defender:defenderOwner,targetLabel:locationLabel(targetId),
      attackerUnitsText:`⚔ ${legions}${cavalry?` · 🐎 ${cavalry}`:''}`,
      attackerBefore:attackUnits,attackerLosses,attackerSurvivors,
      defenderBefore,defenderLosses,defenderAfter,
      attackPower:atkPower,defensePower:defPower,attackerWon,captured
    });
    if(captured&&node?.type==='rome') toast('🏛️ Roma kalesi alındı! Bağlantı korunursa sonraki tur Roma Coin üretir.');
    else toast(attackerWon?'Saldırı başarılı!':'Savunma saldırıyı durdurdu.',attackerWon);
  };
  resolveBattle = function(sourceId,targetId,committed,atkSupport='none',defSupport='none') { return resolveMixedBattle(sourceId,targetId,committed,0,atkSupport,defSupport); };

  // Speed upgrades also affect automatic march orders.
  if (typeof processOneAutoMarch === 'function') {
    processOneAutoMarch = function(order) {
      if(order.owner!==game.current||order.status==='done'||game.turnSerial<=order.createdTurnSerial)return false;
      const g=troopGroup(order.currentLocationId); if(!g||g.owner!==order.owner){order.status='paused';return false;}
      const legions=Math.min(order.legions,g.ready||0), cavalry=Math.min(order.cavalry,cavalryReady(g));
      if(legions+cavalry<1){order.status='paused';return false;}
      order.legions=legions;order.cavalry=cavalry;
      if(order.currentLocationId===order.destinationId){order.status='done';return false;}
      const path=autoShortestPath(order.currentLocationId,order.destinationId,order.owner); if(!path||path.length<2){order.status='paused';return false;}
      let speed=selectedArmySpeed(order.owner,legions,cavalry), targetIndex=Math.min(speed,path.length-1);
      const oldCurrent=game.current; game.current=order.owner;
      for(let i=1;i<=targetIndex;i++){
        if(targetIsHostileForCurrentPlayer(path[i])){targetIndex=i-1;order.status='waiting-battle';break;}
      }
      game.current=oldCurrent; if(targetIndex<1)return false;
      const targetId=path[targetIndex]; game.current=order.owner; autoMarchExecuting=true;
      const moved=moveMixedFriendly(order.currentLocationId,targetId,legions,cavalry); autoMarchExecuting=false; game.current=oldCurrent;
      if(!moved){order.status='paused';return false;}
      order.currentLocationId=targetId; order.status=targetId===order.destinationId?'done':'active';
      addLog(`otomatik yürüdü: ${locationLabel(targetId)} · Hız ${speed}.`,PLAYER_NAMES[order.owner]); return true;
    };
  }
