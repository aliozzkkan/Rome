  function cavalryReady(g) { return g ? (g.cavReady || 0) : 0; }
  function cavalrySpent(g) { return g ? (g.cavSpent || 0) : 0; }
  function cavalryTotalInGroup(g) { return cavalryReady(g) + cavalrySpent(g); }
  function legionTotalInGroup(g) { return g ? (g.ready || 0) + (g.spent || 0) : 0; }
  function unitTotalInGroup(g) { return legionTotalInGroup(g) + cavalryTotalInGroup(g); }
  function unitReadyTotal(g) { return g ? (g.ready || 0) + cavalryReady(g) : 0; }
  function unitPowerInGroup(g) {
    if (!g) return 0;
    return legionTotalInGroup(g) + cavalryTotalInGroup(g) * 2;
  }

  for (const g of troopGroups.values()) {
    if (g.cavReady == null) g.cavReady = 0;
    if (g.cavSpent == null) g.cavSpent = 0;
  }

  troopTotal = function(locationId) {
    return unitTotalInGroup(troopGroup(locationId));
  };

  setTroops = function(locationId, owner, ready, spent = 0) {
    const old = troopGroups.get(locationId);
    const keepCavalry = old && old.owner === owner;
    const cavReadyValue = keepCavalry ? cavalryReady(old) : 0;
    const cavSpentValue = keepCavalry ? cavalrySpent(old) : 0;
    if (ready + spent + cavReadyValue + cavSpentValue <= 0) {
      troopGroups.delete(locationId);
    } else {
      troopGroups.set(locationId, {
        owner,
        ready,
        spent,
        cavReady: cavReadyValue,
        cavSpent: cavSpentValue,
      });
    }
    syncTroopVisual(locationId);
  };

  addTroops = function(locationId, owner, readyAdd = 0, spentAdd = 0) {
    let g = troopGroups.get(locationId);
    if (g && g.owner !== owner) return false;
    if (!g) {
      g = { owner, ready: 0, spent: 0, cavReady: 0, cavSpent: 0 };
      troopGroups.set(locationId, g);
    }
    g.ready = (g.ready || 0) + readyAdd;
    g.spent = (g.spent || 0) + spentAdd;
    if (g.cavReady == null) g.cavReady = 0;
    if (g.cavSpent == null) g.cavSpent = 0;
    syncTroopVisual(locationId);
    return true;
  };

  function addCavalry(locationId, owner, readyAdd = 0, spentAdd = 0) {
    let g = troopGroups.get(locationId);
    if (g && g.owner !== owner) return false;
    if (!g) {
      g = { owner, ready: 0, spent: 0, cavReady: 0, cavSpent: 0 };
      troopGroups.set(locationId, g);
    }
    if (g.cavReady == null) g.cavReady = 0;
    if (g.cavSpent == null) g.cavSpent = 0;
    g.cavReady += readyAdd;
    g.cavSpent += spentAdd;
    syncTroopVisual(locationId);
    return true;
  }

  removeTroops = function(locationId, readyRemove = 0, totalRemove = 0) {
    const g = troopGroups.get(locationId);
    if (!g) return;
    if (readyRemove) g.ready = Math.max(0, (g.ready || 0) - readyRemove);
    if (totalRemove) {
      let left = totalRemove;
      const fromReady = Math.min(g.ready || 0, left);
      g.ready = (g.ready || 0) - fromReady;
      left -= fromReady;
      if (left) g.spent = Math.max(0, (g.spent || 0) - left);
    }
    if (unitTotalInGroup(g) <= 0) troopGroups.delete(locationId);
    syncTroopVisual(locationId);
  };

  function removeCavalry(locationId, readyRemove = 0, totalRemove = 0) {
    const g = troopGroups.get(locationId);
    if (!g) return;
    if (g.cavReady == null) g.cavReady = 0;
    if (g.cavSpent == null) g.cavSpent = 0;
    if (readyRemove) g.cavReady = Math.max(0, g.cavReady - readyRemove);
    if (totalRemove) {
      let left = totalRemove;
      const fromReady = Math.min(g.cavReady, left);
      g.cavReady -= fromReady;
      left -= fromReady;
      if (left) g.cavSpent = Math.max(0, g.cavSpent - left);
    }
    if (unitTotalInGroup(g) <= 0) troopGroups.delete(locationId);
    syncTroopVisual(locationId);
  }

  freeCapacity = function(locationId, owner = game.current) {
    const cap = locationCapacity(locationId);
    if (!Number.isFinite(cap)) return Infinity;
    const g = troopGroup(locationId);
    const occupied = g && g.owner === owner ? unitTotalInGroup(g) : 0;
    return Math.max(0, cap - occupied);
  };

  totalArmies = function(playerId) {
    let n = 0;
    for (const g of troopGroups.values()) if (g.owner === playerId) n += unitTotalInGroup(g);
    return n;
  };

  refreshLegionsForTurn = function(playerId) {
    for (const [locId, g] of troopGroups) {
      if (g.owner !== playerId) continue;
      g.ready = (g.ready || 0) + (g.spent || 0);
      g.spent = 0;
      g.cavReady = cavalryReady(g) + cavalrySpent(g);
      g.cavSpent = 0;
      syncTroopVisual(locId);
    }
  };

  function makeCavalrySoldier(owner, spent = false) {
    const group = new THREE.Group();
    const color = troopColor(owner);
    const opacity = spent ? .52 : 1;
    const horseMat = new THREE.MeshStandardMaterial({ color: 0x6b4932, roughness: .88, transparent: spent, opacity });
    const riderMat = new THREE.MeshStandardMaterial({ color, roughness: .62, transparent: spent, opacity });
    const metalMat = new THREE.MeshStandardMaterial({ color: 0xd8d0c0, roughness: .4, metalness: .5, transparent: spent, opacity });

    const body = new THREE.Mesh(new THREE.CapsuleGeometry(.09, .28, 5, 10), horseMat);
    body.rotation.z = Math.PI / 2;
    body.position.y = .17;
    body.castShadow = true;
    group.add(body);

    const neck = new THREE.Mesh(new THREE.CylinderGeometry(.04, .055, .19, 8), horseMat);
    neck.rotation.z = -.48;
    neck.position.set(.18, .29, 0);
    group.add(neck);

    const head = new THREE.Mesh(new THREE.BoxGeometry(.11,.09,.08), horseMat);
    head.position.set(.24,.38,0);
    group.add(head);

    for (const x of [-.12,.12]) {
      for (const z of [-.055,.055]) {
        const leg = new THREE.Mesh(new THREE.CylinderGeometry(.016,.018,.24,7), horseMat);
        leg.position.set(x,.02,z);
        group.add(leg);
      }
    }

    const rider = new THREE.Mesh(new THREE.CylinderGeometry(.055,.07,.2,9), riderMat);
    rider.position.set(-.02,.39,0);
    group.add(rider);

    const riderHead = new THREE.Mesh(new THREE.SphereGeometry(.05,10,8), new THREE.MeshStandardMaterial({ color:0xd2aa83, roughness:.85, transparent:spent, opacity }));
    riderHead.position.set(-.02,.53,0);
    group.add(riderHead);

    const helmet = new THREE.Mesh(new THREE.SphereGeometry(.057,10,7,0,Math.PI*2,0,Math.PI*.55), metalMat);
    helmet.position.set(-.02,.55,0);
    group.add(helmet);

    const spear = new THREE.Mesh(new THREE.CylinderGeometry(.009,.009,.48,7), metalMat);
    spear.rotation.z = -.72;
    spear.position.set(.1,.47,-.05);
    group.add(spear);

    group.scale.setScalar(.9);
    return group;
  }

  function makeMixedTroopStackVisual(owner, g, moving = false) {
    const group = new THREE.Group();
    group.userData.troopStack = true;
    const lReady = g?.ready || 0;
    const lSpent = g?.spent || 0;
    const cReady = cavalryReady(g);
    const cSpent = cavalrySpent(g);
    const spots = [[-.23,-.18],[.23,-.18],[-.23,.18],[.23,.18],[0,0]];
    let idx = 0;

    const cavShown = Math.min(cReady + cSpent, 2);
    for (let i=0;i<cavShown && idx<spots.length;i++,idx++) {
      const spent = !moving && i >= cReady;
      const unit = makeCavalrySoldier(owner, spent);
      unit.position.set(spots[idx][0], 0, spots[idx][1]);
      group.add(unit);
    }

    const legionShown = Math.min(lReady + lSpent, spots.length - idx);
    for (let i=0;i<legionShown && idx<spots.length;i++,idx++) {
      const spent = !moving && i >= lReady;
      const unit = makeLegionSoldier(owner, spent);
      unit.position.set(spots[idx][0], 0, spots[idx][1]);
      group.add(unit);
    }

    const ring = new THREE.Mesh(
      new THREE.CylinderGeometry(.4,.4,.025,28),
      new THREE.MeshStandardMaterial({ color:troopColor(owner), roughness:.75, transparent:true, opacity:.42 })
    );
    ring.position.y = -.095;
    group.add(ring);

    const labelText = owner
      ? `P${owner}  ⚔${lReady+lSpent}${cReady+cSpent ? `  🐎${cReady+cSpent}` : ''}`
      : `ROMA ⚔${lReady+lSpent}`;
    const label = makeLabel(labelText, owner ? (PLAYER_CSS[owner] || '#f6eddc') : '#e7ddd0', .43);
    label.position.set(0,.82,0);
    group.add(label);
    return group;
  }

  syncTroopVisual = function(locationId) {
    clearTroopVisual(locationId);
    const node = nodes.get(locationId);
    if (node) {
      node.armies = troopTotal(locationId);
      updateNodeVisual(node);
    }
    const g = troopGroup(locationId);
    if (!g) return;
    const visual = makeMixedTroopStackVisual(g.owner,g,false);
    visual.position.copy(locationWorldPosition(locationId));
    scene.add(visual);
    troopVisuals.set(locationId,visual);
  };

  function animateMixedMarch(owner, legions, cavalry, sourceId, targetId, kind = 'move') {
    const start = locationWorldPosition(sourceId);
    const end = locationWorldPosition(targetId);
    const tempGroup = {
      owner,
      ready: Math.min(legions,3),
      spent: 0,
      cavReady: Math.min(cavalry,2),
      cavSpent: 0
    };
    const visual = makeMixedTroopStackVisual(owner,tempGroup,true);
    visual.position.copy(start);
    const dx = end.x-start.x;
    const dz = end.z-start.z;
    visual.rotation.y = Math.atan2(dx,dz);
    scene.add(visual);
    troopAnimations.push({
      visual,start,end,startedAt:performance.now(),
      duration: kind === 'battle' ? 520 : (cavalry && !legions ? 520 : 700),
      kind
    });
  }

  const baseSelectCardForCavalry = selectCard;
  selectCard = function(cardId) {
    if (game.gameOver || game.phase !== 'play' || currentPlayer().isBot) return;
    const p = currentPlayer();
    const c = p.hand.find(x => x.id === cardId);
    if (c?.type === 'reinforcement') {
      if (!canPlayAnotherCard()) { toast('Bu tur en fazla 3 kart oynayabilirsin.', false); return; }
      if (p.hand.length <= 1) { toast('Turu bitirmek için en az 1 kartı elinde bırakmalısın.', false); return; }
      game.selectedCardId = cardId;
      playReinforcement();
      return;
    }
    return baseSelectCardForCavalry(cardId);
  };

  playReinforcement = function() {
    if (game.phase !== 'play') return;
    if (!canPlayAnotherCard()) { toast('Bu tur 3 kart sınırına ulaştın.', false); return; }
    if (currentPlayer().hand.length <= 1) { toast('Son kartını oynayamazsın; tur sonunda 1 kart atmalısın.', false); return; }
    const c = currentPlayer().hand.find(x => x.id === game.selectedCardId && x.type === 'reinforcement');
    if (!c) return;
    const homeId = `home-${game.current}`;
    if (freeCapacity(homeId,game.current) < 1) {
      toast(`Ana kale dolu (${troopTotal(homeId)}/${locationCapacity(homeId)}).`, false);
      game.selectedCardId = null;
      renderAll();
      return;
    }
    addCavalry(homeId,game.current,0,1);
    consumeSelectedCard(1);
    addLog('ana kalede 1 Atlı Birlik çıkardı. Güç 2, hareket 2; bu tur bekler.', PLAYER_NAMES[game.current]);
    toast('🐎 Atlı Birlik ana kalede doğdu. Gelecek tur 2 kart ilerleyebilir.');
    resetInteractionMode();
  };

  function completedReachableWithin(sourceId, maxDistance) {
    const result = new Map();
    const q = [{id:sourceId,d:0}];
    const seen = new Map([[sourceId,0]]);
    while (q.length) {
      const {id,d} = q.shift();
      if (d >= maxDistance) continue;
      for (const next of adjacentLocations(id)) {
        const nd = d + 1;
        if (!result.has(next) || nd < result.get(next)) result.set(next,nd);
        const hostile = targetIsHostileForCurrentPlayer(next);
        if (hostile) continue;
        if (!seen.has(next) || nd < seen.get(next)) {
          seen.set(next,nd);
          q.push({id:next,d:nd});
        }
      }
    }
    result.delete(sourceId);
    return result;
  }

  function movementDistance(sourceId,targetId,maxDistance=2) {
    return completedReachableWithin(sourceId,maxDistance).get(targetId) || Infinity;
  }

  function updateMixedPlannerReachable() {
    if (!touchArmyPlan) return;
    const hasLegion = (touchArmyPlan.legionCount || 0) > 0;
    const hasCavalry = (touchArmyPlan.cavalryCount || 0) > 0;
    let reachable = new Set();
    if (hasLegion) {
      reachable = adjacentLocations(touchArmyPlan.sourceId);
    } else if (hasCavalry) {
      reachable = new Set(completedReachableWithin(touchArmyPlan.sourceId,2).keys());
    }
    game.reachable = reachable;
    if (touchArmyPlan.targetId && !reachable.has(touchArmyPlan.targetId)) touchArmyPlan.targetId = null;
  }

  function mixedMoveCapacityOkay(targetId,legions,cavalry) {
    if (targetIsHostileForCurrentPlayer(targetId)) return true;
    return freeCapacity(targetId,game.current) >= legions + cavalry;
  }

  function moveMixedFriendly(sourceId,targetId,legions,cavalry) {
    const sg = troopGroup(sourceId);
    if (!sg || sg.owner !== game.current) return false;
    if ((sg.ready || 0) < legions || cavalryReady(sg) < cavalry) return false;
    const dist = movementDistance(sourceId,targetId,2);
    const allowed = legions > 0 ? dist === 1 : (cavalry > 0 && dist <= 2);
    if (!allowed) { toast('Lejyon 1 kart, Atlı Birlik 2 kart hareket edebilir.', false); return false; }
    if (!mixedMoveCapacityOkay(targetId,legions,cavalry)) {
      toast(`${locationLabel(targetId)} kapasitesi yetersiz.`, false);
      return false;
    }

    removeTroops(sourceId,legions,0);
    removeCavalry(sourceId,cavalry,0);
    animateMixedMarch(game.current,legions,cavalry,sourceId,targetId,'move');

    let tg = troopGroup(targetId);
    if (!tg) {
      tg = { owner:game.current, ready:0, spent:0, cavReady:0, cavSpent:0 };
      troopGroups.set(targetId,tg);
    }
    if (tg.owner !== game.current) return false;
    tg.spent = (tg.spent || 0) + legions;
    tg.cavSpent = cavalrySpent(tg) + cavalry;

    const node = nodes.get(targetId);
    if (node && node.type !== 'home' && (node.owner === 0 || node.owner === game.current)) node.owner = game.current;
    syncTroopVisual(sourceId);
    syncTroopVisual(targetId);
    addLog(`${legions ? `${legions} Lejyon` : ''}${legions && cavalry ? ' + ' : ''}${cavalry ? `${cavalry} Atlı` : ''} ilerledi (${dist} kart).`, PLAYER_NAMES[game.current]);
    return true;
  }

  function removeDefenderUnits(g,count) {
    if (!g || count <= 0) return;
    let left = count;
    const lr = Math.min(g.ready || 0,left); g.ready = (g.ready || 0)-lr; left -= lr;
    const ls = Math.min(g.spent || 0,left); g.spent = (g.spent || 0)-ls; left -= ls;
    const cr = Math.min(cavalryReady(g),left); g.cavReady = cavalryReady(g)-cr; left -= cr;
    const cs = Math.min(cavalrySpent(g),left); g.cavSpent = cavalrySpent(g)-cs;
  }

  function resolveMixedBattle(sourceId,targetId,legions,cavalry,atkSupport='none',defSupport='none') {
    const attacker = game.current;
    const source = troopGroup(sourceId);
    const target = troopGroup(targetId);
    const node = nodes.get(targetId);
    if (!source || source.owner !== attacker || (source.ready || 0) < legions || cavalryReady(source) < cavalry) return;

    const dist = movementDistance(sourceId,targetId,2);
    if (legions > 0 && dist !== 1) return;
    if (legions === 0 && cavalry > 0 && dist > 2) return;

    const defenderOwner = target?.owner || node?.owner || 0;
    const atkSupportPower = supportPower(atkSupport,'atk',node);
    const defSupportPower = supportPower(defSupport,'def',node);
    if (atkSupportPower && atkSupport !== 'none') removeOneSupportCard(attacker,atkSupport,true);
    if (defSupportPower && defSupport !== 'none' && defenderOwner) removeOneSupportCard(defenderOwner,defSupport,false);

    const attackUnits = legions + cavalry;
    const atkPower = legions + cavalry*2 + atkSupportPower;
    const defenderUnits = unitTotalInGroup(target);
    const defPower = unitPowerInGroup(target) + (node ? defenseBonus(node) : 0) + defSupportPower;

    removeTroops(sourceId,legions,0);
    removeCavalry(sourceId,cavalry,0);
    animateMixedMarch(attacker,legions,cavalry,sourceId,targetId,'battle');

    if (atkPower > defPower) {
      let casualties = Math.min(Math.max(0,Math.floor(defenderUnits/2)), Math.max(0,attackUnits-1));
      let surviveLegion = legions;
      let surviveCavalry = cavalry;
      const lLoss = Math.min(surviveLegion,casualties);
      surviveLegion -= lLoss;
      casualties -= lLoss;
      surviveCavalry = Math.max(0,surviveCavalry-casualties);
      if (surviveLegion + surviveCavalry < 1) {
        if (cavalry > 0) surviveCavalry = 1;
        else surviveLegion = 1;
      }

      troopGroups.delete(targetId);

      if (node?.type === 'home') {
        const oldOwner = node.owner;
        const homeLegion = Math.max(1, Math.ceil((target?.ready || target?.spent || 1)/2));
        troopGroups.set(targetId,{owner:oldOwner,ready:homeLegion,spent:0,cavReady:0,cavSpent:0});
        let back = troopGroups.get(sourceId);
        if (!back) {
          back = {owner:attacker,ready:0,spent:0,cavReady:0,cavSpent:0};
          troopGroups.set(sourceId,back);
        }
        back.spent += surviveLegion;
        back.cavSpent = cavalrySpent(back)+surviveCavalry;
        const stolen = stealRandomCard(oldOwner,attacker);
        addLog(`ana kaleyi bastı; ${stolen ? '1 kart ganimet aldı' : 'ganimet bulamadı'}.`,PLAYER_NAMES[attacker]);
      } else {
        const cap = node && Number.isFinite(nodeCapacity(node)) ? nodeCapacity(node) : Infinity;
        let allowed = Math.min(surviveLegion+surviveCavalry,cap);
        const placeCav = Math.min(surviveCavalry,allowed);
        allowed -= placeCav;
        const placeLegion = Math.min(surviveLegion,allowed);
        troopGroups.set(targetId,{owner:attacker,ready:0,spent:placeLegion,cavReady:0,cavSpent:placeCav});
        if (node) {
          node.owner = attacker;
          addLog(node.type === 'rome' ? `${locationLabel(targetId)} ele geçirildi!` : 'bir kaleyi ele geçirdi.',PLAYER_NAMES[attacker]);
        } else {
          addLog('yol savaşını kazandı.',PLAYER_NAMES[attacker]);
        }
      }
      toast(node?.type === 'rome' ? '🏛️ Roma kalesi ele geçirildi!' : 'Saldırı başarılı!');
    } else {
      const defenderLoss = Math.min(defenderUnits,Math.floor(attackUnits/2));
      if (target && defenderLoss > 0) {
        removeDefenderUnits(target,defenderLoss);
        if (unitTotalInGroup(target) <= 0) troopGroups.delete(targetId);
      }
      addLog(`saldırıda durduruldu (${atkPower} vs ${defPower}).`,PLAYER_NAMES[attacker]);
      toast('Savunma saldırıyı durdurdu.',false);
    }
    syncTroopVisual(sourceId);
    syncTroopVisual(targetId);
    renderAll();
  }

  resolveBattle = function(sourceId,targetId,committed,atkSupport='none',defSupport='none') {
    return resolveMixedBattle(sourceId,targetId,committed,0,atkSupport,defSupport);
  };

  openTouchArmyPlanner = function(sourceId) {
    if (game.gameOver || game.phase !== 'play' || currentPlayer().isBot) return;
    const g = troopGroup(sourceId);
    if (!g || g.owner !== game.current || unitReadyTotal(g) < 1) {
      toast('Burada bu tur hareket edebilecek birliğin yok.',false);
      return;
    }
    touchArmyPlan = {
      sourceId,
      targetId:null,
      legionCount:(g.ready||0)>0 ? 1 : 0,
      cavalryCount:(g.ready||0)>0 ? 0 : 1
    };
    game.selectedCardId = null;
    game.selectedLocationId = sourceId;
    game.selectedNodeId = nodes.has(sourceId) ? sourceId : null;
    game.mode = 'move-target';
    updateMixedPlannerReachable();
    if (!game.reachable.size) {
      hideTouchArmyPlanner();
      toast('Bu birlik tamamlanmış bir yola bağlı değil.',false);
      return;
    }
    setModeText('ORDU: Lejyon 1 kart, Atlı Birlik 2 kart ilerler. Birlik sayılarını seç ve hedefe dokun.');
    renderAll();
    renderTouchArmyPlanner();
  };

    if (!g || g.owner !== game.current || unitReadyTotal(g)<1) { hideTouchArmyPlanner(); return; }

    touchArmyPlan.cavalryCount = Math.max(0,Math.min(cMax,touchArmyPlan.cavalryCount||0));
    if (touchArmyPlan.legionCount + touchArmyPlan.cavalryCount < 1) {
      if (lMax) touchArmyPlan.legionCount = 1;
      else if (cMax) touchArmyPlan.cavalryCount = 1;
    }
    updateMixedPlannerReachable();

    const targetId = touchArmyPlan.targetId;
    const targetText = targetId
      ? `<strong>${locationLabel(targetId)}</strong> · ${targetIsHostileForCurrentPlayer(targetId) ? '⚔️ saldırı' : `${movementDistance(touchArmyPlan.sourceId,targetId,2)} kart hareket`}`
      : '<strong>Hedef seç</strong> · parlayan kareye dokun';

    touchArmyPanel.innerHTML = `
      <div class="tap-head">
        <div>
          <div class="tap-title">${targetId && typeof isRomeCastleId === 'function' && isRomeCastleId(targetId) ? 'ROMA KALESİNE SALDIRI' : 'ORDU HAREKETİ'}</div>
          <div class="tap-route">${locationLabel(touchArmyPlan.sourceId)} · ⚔ ${lMax} hazır · 🐎 ${cMax} hazır</div>
        </div>
        <button id="tapArmyClose">✕</button>
      </div>

    touchArmyPanel.classList.add('show');

  selectTouchArmyTarget = function(targetId) {
    if (!touchArmyPlan) return;
    updateMixedPlannerReachable();
    if (!game.reachable.has(targetId)) {
      toast('Bu birlik seçimiyle o hedefe bu tur ulaşılamıyor.',false);
      return;
    }
    const l = touchArmyPlan.legionCount || 0;
    const c = touchArmyPlan.cavalryCount || 0;
    if (!mixedMoveCapacityOkay(targetId,l,c)) {
      toast(`${locationLabel(targetId)} kapasitesi yetersiz.`,false);
      return;
    }
    touchArmyPlan.targetId = targetId;
    renderAll();
    renderTouchArmyPlanner();
  };

  function openMixedBattleConfirm(sourceId,targetId,legions,cavalry) {
    const tg = troopGroup(targetId);
    const defenderOwner = tg?.owner || nodes.get(targetId)?.owner || 0;
    const node = nodes.get(targetId);
    const defBonus = node ? defenseBonus(node) : 0;
    const atkT = supportCount(game.current,'tactic');
    const atkS = node && (node.type === 'fort' || node.type === 'rome') ? supportCount(game.current,'siege') : 0;
    const defT = defenderOwner ? supportCount(defenderOwner,'tactic') : 0;
    const defS = defenderOwner ? supportCount(defenderOwner,'shield') : 0;
    const atkPower = legions + cavalry*2;

    modalEl.innerHTML = `
      <h3>⚔️ Saldırı</h3>
      <p>${locationLabel(sourceId)} → ${locationLabel(targetId)}</p>
      <div class="modalRow"><span>Gönderilen</span><strong>⚔ ${legions} + 🐎 ${cavalry} = ${atkPower} temel güç</strong></div>
      <div class="modalRow"><label for="atkSupport">Saldırı desteği</label><select id="atkSupport" style="max-width:250px"><option value="none">Yok</option>${atkT ? `<option value="tactic">🦅 Taktik +2 (${atkT})</option>` : ''}${atkS ? `<option value="siege">🧨 Kuşatma +3 (${atkS})</option>` : ''}</select></div>
      <div class="modalRow"><label for="defSupport">Savunma desteği</label><select id="defSupport" style="max-width:250px"><option value="none">Yok</option>${defT ? `<option value="tactic">🦅 Taktik +2 (${defT})</option>` : ''}${defS ? `<option value="shield">🛡️ Kalkan +3 (${defS})</option>` : ''}</select></div>
      <div class="modalRow"><span>Savunan güç</span><strong>${unitPowerInGroup(tg)} birlik gücü + ${defBonus} kale</strong></div>
      <div class="modalActions"><button id="modalCancel">Vazgeç</button><button id="modalConfirm" class="primary">Saldırıyı Gönder</button></div>`;

    modalShade.classList.add('show');
    document.getElementById('modalCancel').addEventListener('click',()=>{ closeModal(); openTouchArmyPlanner(sourceId); });
    document.getElementById('modalConfirm').addEventListener('click',()=>{
      resolveMixedBattle(sourceId,targetId,legions,cavalry,document.getElementById('atkSupport').value,document.getElementById('defSupport').value);
      closeModal(); hideTouchArmyPlanner(); resetInteractionMode();
    });
  }

  sendTouchArmyPlan = function() {
    if (!touchArmyPlan?.targetId) return;
    const {sourceId,targetId} = touchArmyPlan;
    const legions = touchArmyPlan.legionCount || 0;
    const cavalry = touchArmyPlan.cavalryCount || 0;
    if (legions + cavalry < 1) return;
    const sg = troopGroup(sourceId);
    if (!sg || sg.owner !== game.current || (sg.ready||0)<legions || cavalryReady(sg)<cavalry) {
      toast('Birlik durumu değişti; yeniden seç.',false);
      cancelTouchArmyPlanner();
      return;
    }
    updateMixedPlannerReachable();
    if (!game.reachable.has(targetId)) {
      toast('Bu hedef artık ulaşılabilir değil.',false);
      return;
    }
    if (targetIsHostileForCurrentPlayer(targetId)) {
      hideTouchArmyPlanner();
      openMixedBattleConfirm(sourceId,targetId,legions,cavalry);
      return;
    }
    if (moveMixedFriendly(sourceId,targetId,legions,cavalry)) {
      hideTouchArmyPlanner();
      resetInteractionMode();
    }
  };

  if (typeof readyRomeCastleAttacks === 'function') {
    readyRomeCastleAttacks = function() {
      if (game.gameOver || game.phase !== 'play' || currentPlayer().isBot) return [];
      const pairs = [];
      for (const [sourceId,g] of troopGroups.entries()) {
        if (g.owner !== game.current || unitReadyTotal(g)<1) continue;
        const maxDist = cavalryReady(g)>0 ? 2 : 1;
        const reach = completedReachableWithin(sourceId,maxDist);
        for (const [targetId,dist] of reach.entries()) {
          if (!isRomeCastleId(targetId) || !targetIsHostileForCurrentPlayer(targetId)) continue;
          if (dist > 1 && cavalryReady(g) < 1) continue;
          pairs.push({sourceId,targetId});
        }
      }
      return pairs;
    };

    beginRomeCastleAttack = function(sourceId,targetId) {
      const g = troopGroup(sourceId);
      if (!g || g.owner !== game.current || unitReadyTotal(g)<1) return;
      openTouchArmyPlanner(sourceId);
      if (!touchArmyPlan) return;
      const targetDistance = movementDistance(sourceId,targetId,2);
      if (targetDistance > 1 && cavalryReady(g)>0) {
        touchArmyPlan.legionCount = 0;
        touchArmyPlan.cavalryCount = 1;
        updateMixedPlannerReachable();
      } else if (!game.reachable.has(targetId) && cavalryReady(g)>0) {
        touchArmyPlan.legionCount = 0;
        touchArmyPlan.cavalryCount = 1;
        updateMixedPlannerReachable();
      }
      if (!game.reachable.has(targetId)) {
        toast('Bu birlik seçimiyle Roma kalesine bu tur ulaşılamıyor.',false);
        return;
      }
      touchArmyPlan.targetId = targetId;
      renderAll();
      renderTouchArmyPlanner();
    };
  }

  const baseBotPlayableOptionsCavalry = botPlayableOptions;
  botPlayableOptions = function(playerId) {
    const options = baseBotPlayableOptionsCavalry(playerId).filter(o => o.type !== 'reinforcement');
    const p = playerById(playerId);
    if (p && p.hand.length > 1 && canPlayAnotherCard() && p.hand.some(c=>c.type==='reinforcement') && freeCapacity(`home-${playerId}`,playerId)>=1) {
      options.push({type:'reinforcement',candidates:null});
    }
    return options;
  };

  botMoveOnce = function(playerId) {
    const sources = [...troopGroups.entries()]
      .filter(([,g])=>g.owner===playerId && ((g.ready||0)>0 || cavalryReady(g)>0))
      .map(([id])=>id);
    const sourceId = randomChoice(sources);
    if (!sourceId) return false;
    const g = troopGroup(sourceId);
    const useCavalry = cavalryReady(g)>0 && ((g.ready||0)===0 || Math.random()<.45);
    const reachMap = completedReachableWithin(sourceId,useCavalry?2:1);
    const targets = [...reachMap.keys()];
    const targetId = randomChoice(targets);
    if (!targetId) return false;
    const count = randomInt(1,useCavalry?cavalryReady(g):(g.ready||0));
    const hostile = targetIsHostileForCurrentPlayer(targetId);
    const oldCurrent = game.current;
    game.current = playerId;
    if (hostile) resolveMixedBattle(sourceId,targetId,useCavalry?0:count,useCavalry?count:0,'none','none');
    else moveMixedFriendly(sourceId,targetId,useCavalry?0:count,useCavalry?count:0);
    game.current = oldCurrent;
    return true;
  };

  renderer.domElement.addEventListener('dblclick', e => {
    if (game.gameOver || game.phase !== 'play' || currentPlayer().isBot) return;
    const loc = locationUnderPointer(e);
    const g = loc ? troopGroup(loc.id) : null;
    if (loc && g && g.owner === game.current && unitReadyTotal(g)>0) openTouchArmyPlanner(loc.id);
  });

  let cavTouchTap = {id:null,time:0};
  let cavTouchPress = null;
  renderer.domElement.addEventListener('pointerdown', e => {
    if (e.pointerType !== 'touch' || game.gameOver || game.phase !== 'play' || currentPlayer().isBot || touchArmyPlan) return;
    const loc = locationUnderPointer(e);
    const g = loc ? troopGroup(loc.id) : null;
    if (!loc || !g || g.owner !== game.current || cavalryReady(g)<1 || (g.ready||0)>0) return;
    cavTouchPress = {id:loc.id,x:e.clientX,y:e.clientY};
    e.preventDefault();
    e.stopImmediatePropagation();
  }, {capture:true,passive:false});

  renderer.domElement.addEventListener('pointerup', e => {
    if (e.pointerType !== 'touch' || !cavTouchPress) return;
    const press = cavTouchPress;
    cavTouchPress = null;
    e.preventDefault();
    e.stopImmediatePropagation();
    if (Math.hypot(e.clientX-press.x,e.clientY-press.y)>16) return;
    const loc = locationUnderPointer(e);
    if (!loc || loc.id!==press.id) return;
    const now = performance.now();
    const isDouble = cavTouchTap.id===loc.id && now-cavTouchTap.time<480;
    cavTouchTap = {id:loc.id,time:now};
    if (isDouble) {
      cavTouchTap = {id:null,time:0};
      openTouchArmyPlanner(loc.id);
    } else {
      toast(`🐎 ${cavalryReady(troopGroup(loc.id))} hazır Atlı. Tekrar dokunarak birlik panelini aç.`,true);
    }
  }, {capture:true,passive:false});
