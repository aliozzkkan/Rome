
  // === PHASE 1: automatic T / + junctions when a straight road passes a castle ===
  const baseDynamicRoadConnects = roadConnects;

  function junctionNodeBesideRoad(r, dirIndex) {
    if (!r) return null;
    const d = DIRS[dirIndex];
    const c = cellMap.get(k(r.gx + d.dx, r.gz + d.dz));
    if (!c || c.kind !== 'node') return null;
    const n = nodes.get(c.id);
    if (!n) return null;
    if (!['fort','home','rome'].includes(n.type)) return null;
    return n;
  }

  roadConnects = function(r, dirIndex) {
    if (!r) return false;
    if (baseDynamicRoadConnects(r, dirIndex)) return true;
    // A straight road automatically opens a side spur when a castle is directly beside it.
    return !!junctionNodeBesideRoad(r, dirIndex);
  };

  function clearAutoJunctionVisuals(r) {
    if (!r?.group) return;
    for (const child of [...r.group.children]) {
      if (!child.userData?.autoJunctionBranch) continue;
      r.group.remove(child);
      child.geometry?.dispose?.();
      child.material?.dispose?.();
    }
  }

  function makeAutoJunctionBranch(r, dirIndex) {
    const mat = new THREE.MeshStandardMaterial({ color: 0xd7c7a2, roughness: .82 });
    const mesh = new THREE.Mesh(new THREE.BoxGeometry(TILE * .54, .045, ROAD_WIDTH), mat);
    mesh.position.y = .135;
    // The road group is rotated 90° for horizontal roads. Local X is therefore
    // east/west for vertical roads and north/south for horizontal roads.
    let sign = 1;
    if (r.orientation === 'v') sign = dirIndex === 1 ? 1 : -1; // east / west
    else sign = dirIndex === 0 ? 1 : -1; // north / south after group rotation
    mesh.position.x = sign * TILE * .27;
    mesh.castShadow = true;
    mesh.userData.autoJunctionBranch = true;
    r.group.add(mesh);
  }

  function refreshAutomaticJunctions() {
    for (const r of roadPieces.values()) {
      clearAutoJunctionVisuals(r);
      const baseDirs = r.orientation === 'v' ? new Set([0,2]) : new Set([1,3]);
      for (let i=0;i<4;i++) {
        if (baseDirs.has(i)) continue;
        if (junctionNodeBesideRoad(r,i)) makeAutoJunctionBranch(r,i);
      }
    }
  }

  const baseDynamicUpdateRoadCompletionStatus = updateRoadCompletionStatus;
  updateRoadCompletionStatus = function(...args) {
    const result = baseDynamicUpdateRoadCompletionStatus.apply(this,args);
    refreshAutomaticJunctions();
    return result;
  };

  const baseDynamicRenderAll = renderAll;
  renderAll = function(...args) {
    const result = baseDynamicRenderAll.apply(this,args);
    refreshAutomaticJunctions();
    return result;
  };

  requestAnimationFrame(refreshAutomaticJunctions);
