  function locationWorldPosition(locationId) {
    const road = roadPieces.get(locationId);
    if (road) {
      const p = world(road.gx, road.gz);
      return new THREE.Vector3(p.x, .28, p.z);
    }
    const node = nodes.get(locationId);
    if (!node || !node.cells?.length) return new THREE.Vector3();
    let gx = 0, gz = 0;
    for (const c of node.cells) { gx += c.gx; gz += c.gz; }
    gx /= node.cells.length; gz /= node.cells.length;
    const p = world(gx, gz);
    if (node.cells.length === 1) {
      p.x += .39;
      p.z += .39;
    }
    return new THREE.Vector3(p.x, .30, p.z);
  }

  function disposeObject3D(root) {
    root.traverse(obj => {
      if (obj.geometry) obj.geometry.dispose?.();
      if (obj.material) {
        const mats = Array.isArray(obj.material) ? obj.material : [obj.material];
        for (const mat of mats) {
          mat.map?.dispose?.();
          mat.dispose?.();
        }
      }
    });
  }

  function clearTroopVisual(locationId) {
    const visual = troopVisuals.get(locationId);
    if (!visual) return;
    scene.remove(visual);
    disposeObject3D(visual);
    troopVisuals.delete(locationId);
  }

  function animateTroopMarch(owner, count, sourceId, targetId, kind = 'move') {
    const start = locationWorldPosition(sourceId);
    const end = locationWorldPosition(targetId);
    if (!start || !end) return;
    const visual = makeTroopStackVisual(owner, Math.min(count, 5), 0, true);
    visual.scale.setScalar(.92);
    visual.position.copy(start);
    const dx = end.x - start.x;
    const dz = end.z - start.z;
    visual.rotation.y = Math.atan2(dx, dz);
    scene.add(visual);
    troopAnimations.push({
      visual,
      start,
      end,
      startedAt: performance.now(),
      duration: kind === 'battle' ? 520 : 700,
      kind,
    });
  }

  function updateTroopAnimations(now) {
    for (let i = troopAnimations.length - 1; i >= 0; i--) {
      const a = troopAnimations[i];
      const raw = Math.min(1, Math.max(0, (now - a.startedAt) / a.duration));
      const t = raw * raw * (3 - 2 * raw);
      a.visual.position.lerpVectors(a.start, a.end, t);
      a.visual.position.y += Math.sin(raw * Math.PI) * (a.kind === 'battle' ? .34 : .22);
      const bob = Math.sin(raw * Math.PI * 8) * .025;
      a.visual.position.y += bob;
      if (raw >= 1) {
        scene.remove(a.visual);
        disposeObject3D(a.visual);
        troopAnimations.splice(i, 1);
      }
    }
  }

  function updateNodeVisual(node) {
    for (const g of node.groups) {
      const base = g.children[0];
      if (base?.material?.color) base.material.color.setHex(node.type === 'rome' ? (node.owner ? PLAYER_COLORS[node.owner] : 0x8e7444) : PLAYER_COLORS[node.owner]);
      for (const child of [...g.children]) {
        if (child.userData?.nodeLabel) {
          g.remove(child);
          child.material?.map?.dispose?.();
          child.material?.dispose?.();
        }
      }
    }
    const labelGroup = node.groups[0];
    const prefix = node.type === 'rome' ? 'ROMA' : (node.type === 'home' ? `P${node.owner} ANA KALE` : 'KALE');
    const capText = Number.isFinite(nodeCapacity(node)) ? `${node.armies}/${nodeCapacity(node)}` : `${node.armies}`;
    const levelText = node.type === 'rome' ? '' : ` L${node.level}`;
    const spr = makeLabel(`${prefix}${levelText} • ${capText}`);
    spr.position.set(node.type === 'rome' ? CELL * .5 : 0, 1.08, node.type === 'rome' ? CELL * .5 : 0);
    spr.userData.nodeLabel = true;
    labelGroup.add(spr);
  }

  function createNode(id, type, owner, armies, cells, fixed = false) {
    const node = { id, type, owner, armies, cells, fixed, level: 1, groups: [] };
    nodes.set(id, node);
    cells.forEach((cell, index) => {
      const label = type === 'rome' ? (index === 0 ? 'ROMA' : '') : (index === 0 ? (type === 'home' ? `P${owner}` : 'KALE') : '');
      const g = makeNodeCellGroup(type, owner, label, false);
      const p = world(cell.gx, cell.gz);
      g.position.copy(p);
      scene.add(g);
      node.groups.push(g);
      cellMap.set(k(cell.gx,cell.gz), { kind: 'node', id });
      addInteractive(g, { kind: 'node', nodeId: id });
    });
    updateNodeVisual(node);
    updateRoadCompletionStatus();
    return node;
  }

  function destroyRoad(id) {

  }

  function destroyNode(id) {
    const node = nodes.get(id);
    if (!node || node.fixed) return;
    for (const cell of node.cells) cellMap.delete(k(cell.gx,cell.gz));
    for (const g of node.groups) {
      removeInteractiveFor(g);
      scene.remove(g);
    }
    nodes.delete(id);
    updateRoadCompletionStatus();
  }

  function roadConnects(r, dirIndex) {
    if (!r) return false;
    return r.orientation === 'v' ? (dirIndex === 0 || dirIndex === 2) : (dirIndex === 1 || dirIndex === 3);
  }

  function contentAt(gx, gz) {
    const c = cellMap.get(k(gx,gz));
    if (!c) return null;
    return c.kind === 'road' ? roadPieces.get(c.id) : nodes.get(c.id);
  }

  function nodeAt(gx, gz) {
    const c = cellMap.get(k(gx,gz));
    return c?.kind === 'node' ? nodes.get(c.id) : null;
  }

  function roadAt(gx, gz) {
    const c = cellMap.get(k(gx,gz));
    return c?.kind === 'road' ? roadPieces.get(c.id) : null;
  }

  function roadRoadCompatible(orientation, gx, gz, dirIndex, neighbor) {
    const mine = orientation === 'v' ? (dirIndex === 0 || dirIndex === 2) : (dirIndex === 1 || dirIndex === 3);
    const theirs = roadConnects(neighbor, DIRS[dirIndex].opp);
    if (mine !== theirs) return { valid: false, match: false };
    return { valid: true, match: mine && theirs };
  }
